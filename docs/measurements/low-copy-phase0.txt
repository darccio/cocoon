from pathlib import Path
import datetime, hashlib, json, os, re, subprocess

root = Path.cwd()
out = root / '.cache/low-copy'
binary = out / 'baseline.test'
env = dict(os.environ, CGO_ENABLED='0', GOTOOLCHAIN='local', GOAMD64='v1', GOMAXPROCS='16', GOFLAGS='-buildvcs=false')
paths = ['examples/datadog/go/dd/batching_bench_test.go', 'examples/datadog/go/dd/cocoon_gen.go', 'examples/datadog/go/dd/testdata/module.wasm', 'examples/datadog/go/dd/internal/wasm/module.go']
provenance = {
    'base_commit': subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip(),
    'time_utc': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'go': subprocess.check_output(['/usr/lib64/go/1.26/bin/go', 'version'], text=True).strip(),
    'platform': subprocess.check_output(['uname', '-a'], text=True).strip(),
    'affinity': sorted(os.sched_getaffinity(0)),
    'conditions': 'Unpinned Ryzen 75800HS; default GC/memory; CGO0; GOAMD64v1; PGOoff; initial P16; measured leaves assert/report P1 or P16; no compilation/lint/fuzz/reviewer tools concurrently.',
    'sha256': {str(binary.relative_to(root)): hashlib.sha256(binary.read_bytes()).hexdigest()},
    'source_sha256': {p: hashlib.sha256((root / p).read_bytes()).hexdigest() for p in paths},
}
(out / 'phase0-provenance.json').write_text(json.dumps(provenance, indent=2) + '\n')
def sample(label, selector, number):
    args = [str(binary), '-test.run=^$', '-test.bench=' + selector, '-test.benchtime=500ms', '-test.count=1', '-test.benchmem']
    with (out / ('phase0-' + label + '.txt')).open('a') as stream:
        print('ROUND', number, datetime.datetime.now(datetime.timezone.utc).isoformat(), file=stream, flush=True)
        print('COMMAND', json.dumps(args), file=stream, flush=True)
        result = subprocess.run(args, env=env, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, check=True)
        print(result.stdout, file=stream, flush=True)
    for line in result.stdout.splitlines():
        if not line.startswith('Benchmark'): continue
        fields = line.split()
        actual = float(fields[fields.index('actual-procs')-1])
        expected = int(re.search(r'/P(\d+)/', fields[0]).group(1)) if label == 'contention' else 1
        assert actual == expected, line
    print('complete', number, label, flush=True)
for number in range(1, 8):
    modes = ['Scalar', 'Facade', 'Prototype'] if number % 2 else ['Prototype', 'Facade', 'Scalar']
    for mode in modes:
        sample(mode.lower(), '^BenchmarkSketchMatchedBatch$/././^' + mode + '$', number)
    sample('layers', '^BenchmarkSketchBatchLayers$', number)
    sample('contention', '^BenchmarkSketchBatchContention$', number)
print('phase0 complete; seven isolated samples for all controls', flush=True)
