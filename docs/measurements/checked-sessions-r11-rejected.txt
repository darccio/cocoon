		return fmt.Errorf("write checked boundary: %w", err)
	}
	return nil
}

=== internal/gen/rtgen/rtgen.go ===
// Package rtgen renders the shared per-operation runtime safety boundary.
package rtgen

import (
	_ "embed"
	"fmt"
	"go/format"
	"strings"
	"text/template"
)

//go:embed checked.go.tmpl
var checkedTemplate string

// Generate emits locked and caller-held-lock adapters from one safety body.
func Generate() ([]byte, error) {
	source, err := template.New("checked").Parse(checkedTemplate)
	if err != nil {
		return nil, fmt.Errorf("parse checked boundary: %w", err)
	}
	var output strings.Builder
	adapters := []struct {
		Name   string
		Locked bool
	}{{Name: "Call", Locked: true}, {Name: "checked"}}
	if renderErr := source.Execute(&output, adapters); renderErr != nil {
		return nil, fmt.Errorf("render checked boundary: %w", renderErr)
	}
	formatted, err := format.Source([]byte(output.String()))
	if err != nil {
		return nil, fmt.Errorf("format checked boundary: %w", err)
	}
	return formatted, nil
}

package rtgen_test

import (
	"go/ast"
	"go/format"
	"go/parser"
	"go/token"
	"strings"
	"testing"

	"dario.cat/cocoon/internal/gen/rtgen"
)

func TestAdaptersShareEveryCheckedStatement(t *testing.T) {
	t.Parallel()
	source, err := rtgen.Generate()
	if err != nil {
		t.Fatal(err)
	}
	file, err := parser.ParseFile(token.NewFileSet(), "checked_gen.go", source, 0)
	if err != nil {
		t.Fatal(err)
	}
	bodies := make(map[string]string)
	for _, declaration := range file.Decls {
		method, ok := declaration.(*ast.FuncDecl)
		if !ok {
			continue
		}
		if method.Name.Name == "Call" {
			if len(method.Body.List) < 2 {
				t.Fatal("missing locked adapter prefix")
			}
			var prefix strings.Builder
			for _, statement := range method.Body.List[:2] {
				if err := format.Node(&prefix, token.NewFileSet(), statement); err != nil {
					t.Fatal(err)
				}
				prefix.WriteByte('\n')
			}
			if prefix.String() != "i.mu.Lock()\ndefer i.mu.Unlock()\n" {
				t.Fatal("locked adapter must acquire and defer release of execution mutex")
			}
			method.Body.List = method.Body.List[2:]
		}
		var body strings.Builder
		if err := format.Node(&body, token.NewFileSet(), method.Body); err != nil {
			t.Fatal(err)
		}
		bodies[method.Name.Name] = body.String()
	}
	if len(bodies) != 2 || bodies["Call"] == "" || bodies["Call"] != bodies["checked"] {
		t.Fatal("adapters must differ only by the execution lock prefix")
	}
}

	i.mu.Lock()
	defer i.mu.Unlock()
	switch i.state {
	case instanceClosed:
		return ErrClosed
	case instancePoisoned:
		return ErrPoisoned
	case instanceOpen:
	}

		if value := recover(); value != nil {
			i.poison()
			var message string
			if i.host != nil {
				message = i.host.takePanic()
			}
			err = Classify(op, value, message)
		}
	}()
	if i.host != nil {
		i.host.takePanic()
	}

	}
	return err
}
